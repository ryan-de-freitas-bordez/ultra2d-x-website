03 QUICK START

Start with a
clean foundation.

A first look at the Ultra2D-X API style, using the project's current C++ example.

main.cpp
#include <ultra2d.hpp>

USING_NS_UL;

int main()
{
    App::run(ULMain());
}

void ULMain()
{
    if (!ULInit())
        return false;

    auto text = ULLabelBMFont(ULPath("../assets/pusab.fnt"), "Hello World");
    text->setBMFontSize(18);
}
